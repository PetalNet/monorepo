cmake_minimum_required(VERSION 3.16)
project(futo_decode LANGUAGES C CXX)
if(NOT WHISPER_CPP_DIR OR NOT EXISTS "${WHISPER_CPP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set WHISPER_CPP_DIR to a whisper.cpp checkout")
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_CPP_DIR}" whisper-build EXCLUDE_FROM_ALL)
add_executable(futo-decode decode.cpp)
target_compile_features(futo-decode PRIVATE cxx_std_17)
target_link_libraries(futo-decode PRIVATE whisper)
